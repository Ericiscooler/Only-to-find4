import math import random import sys import pygame # Initialize Pygame pygame.init() # Setup Screen Dimensions WIDTH, HEIGHT = 800, 800 screen = pygame.display.set_mode((WIDTH, HEIGHT)) pygame.display.set_caption("Glowing Heart Animation") # Color definitions BLACK = (0, 0, 0) CYAN = (80, 220, 255) GLOW_COLOR = (20, 100, 180) # Heart parametric formula def get_heart_point(t, scale=15): # Standard heart curve x = 16 * (math.sin(t) ** 3) y = -(13 * math.cos(t) - 5 * math.cos(2*t) - 2 * math.cos(3*t) - math.cos(4*t)) return ( int(WIDTH / 2 + x * scale), int(HEIGHT / 2 - 50 + y * scale) ) # Particle class representing each text element inside the heart class HeartParticle: def __init__(self, x, y, delay): self.x = x self.y = y self.delay = delay self.alpha = 0 self.flicker = random.uniform(0.8, 1.2) self.font_size = random.choice([12, 14, 16]) self.font = pygame.font.SysFont("Consolas", self.font_size, bold=True) self.text_surface = self.font.render("Love You", True, CYAN) def update(self, frame_count): if frame_count > self.delay and self.alpha < 255: self.alpha = min(255, self.alpha + 8) if self.alpha >= 255: # Pulsating / flickering wave effect flick = 0.75 + 0.25 * math.sin(frame_count * 0.05 * self.flicker) return min(255, max(0, int(255 * flick))) return self.alpha def draw(self, surface, glow_surface, current_alpha): if current_alpha <= 0: return # Render text with alpha blending text = self.text_surface.copy() text.set_alpha(current_alpha) rect = text.get_rect(center=(self.x, self.y)) # Draw soft outer glow box glow_rect = rect.inflate(12, 8) glow_shape = pygame.Surface((glow_rect.width, glow_rect.height), pygame.SRCALPHA) pygame.draw.rect(glow_shape, (*GLOW_COLOR, min(80, int(current_alpha * 0.3))), glow_shape.get_rect(), border_radius=4) glow_surface.blit(glow_shape, glow_rect.topleft) # Blit main text surface.blit(text, rect) # Build heart particles particles = [] count = 0 # Fill the heart shape by scanning coordinates for scale_step in range(3, 16): points_in_layer = int(scale_step * 12) for i in range(points_in_layer): t = (i / points_in_layer) * 2 * math.pi px, py = get_heart_point(t, scale=scale_step) # Add slight random jitter for a organic layout px += random.randint(-6, 6) py += random.randint(-6, 6) particles.append(HeartParticle(px, py, delay=count * 0.4)) count += 1 # Setup Center Big Text center_font = pygame.font.SysFont("Georgia", 48, bold=True) center_glow_font = pygame.font.SysFont("Georgia", 52, bold=True) clock = pygame.time.Clock() frame = 0 running = True while running: clock.tick(60) frame += 1 for event in pygame.event.get(): if event.type == pygame.QUIT or (event.type == pygame.KEYDOWN and event.key == pygame.K_ESCAPE): running = False screen.fill(BLACK) # Offscreen surface for glow layer glow_layer = pygame.Surface((WIDTH, HEIGHT), pygame.SRCALPHA) text_layer = pygame.Surface((WIDTH, HEIGHT), pygame.SRCALPHA) # Render small heart text particles for p in particles: alpha = p.update(frame) p.draw(text_layer, glow_layer, alpha) # Big "I Love You" reveal after heart fully forms (~frame 180) if frame > 180: pulse = 1.0 + 0.05 * math.sin(frame * 0.08) c_alpha = min(255, (frame - 180) * 5) # Render glowing background text glow_text = center_glow_font.render("I Love You", True, GLOW_COLOR) glow_text.set_alpha(int(c_alpha * 0.6)) # Render main foreground text main_text = center_font.render("I Love You", True, (240, 255, 255)) main_text.set_alpha(c_alpha) # Scale for pulsation w, h = main_text.get_size() scaled_main = pygame.transform.smoothscale(main_text, (int(w * pulse), int(h * pulse))) scaled_glow = pygame.transform.smoothscale(glow_text, (int((w + 20) * pulse), int((h + 20) * pulse))) text_layer.blit(scaled_glow, scaled_glow.get_rect(center=(WIDTH // 2, HEIGHT // 2))) text_layer.blit(scaled_main, scaled_main.get_rect(center=(WIDTH // 2, HEIGHT // 2))) # Composite layers onto screen screen.blit(glow_layer, (0, 0)) screen.blit(text_layer, (0, 0)) pygame.display.flip() pygame.quit() sys.exit()
